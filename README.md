# React UI Kit

Reusable Header and Footer components.

## Install

npm install git+https://openforge.gov.in/plugins/git/yuvaproj/mybharat_common_frontend.git

## Usage

import { Header, Footer } from "mybharat_common_frontend";

function App() {
  return (
    <>
      <Header />
      <Footer />
    </>
  );
}
